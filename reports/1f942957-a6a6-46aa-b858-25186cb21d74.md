## Critical Findings

The following critical findings were identified during the audit:

### 1. VotingEscrow Whitelisting and Unlocking (VE-001)

* Severity: High
* Affected Function: `add_to_whitelist` and `unlock`
* Description: The combination of unrestricted whitelisting and unsecured unlocking in the VotingEscrow contract allows an attacker to manipulate the contract's state.
* Exploit Code:
```solidity
pragma solidity ^0.8.0;
contract VotingEscrow {
    function add_to_whitelist(address _addr) public {
        whitelist[_addr] = true;
    }
    function unlock() public {
        // attacker can call unlock to manipulate state
    }
}
```
* Fix Code:
```solidity
pragma solidity ^0.8.0;
contract VotingEscrow {
    mapping(address => bool) public whitelist;
    address public admin;
    constructor() {
        admin = msg.sender;
    }
    modifier onlyAdmin() {
        require(msg.sender == admin, 'Only admin can call this function');
        _;
    }
    function add_to_whitelist(address _addr) public onlyAdmin {
        whitelist[_addr] = true;
    }
    function unlock() public onlyAdmin {
        // only admin can call unlock
    }
}
```
* Test Code:
```solidity
pragma solidity ^0.8.0;
import 'forge-std/Test.sol';
contract VotingEscrowTest is Test {
    VotingEscrow public votingEscrow;
    address public user1;
    address public user2;
    function setUp() public {
        votingEscrow = new VotingEscrow();
        user1 = address(1);
        user2 = address(2);
    }
    function test_add_to_whitelist() public {
        votingEscrow.add_to_whitelist(user1);
        assertEq(votingEscrow.whitelist(user1), true);
    }
    function test_unlock() public {
        votingEscrow.unlock();
        // should revert if not admin
    }
    function testFail_unlock() public {
        vm.prank(user1);
        votingEscrow.unlock();
        // should revert
    }
}
```

### 2. Unrestricted VotingEscrow Whitelisting (VE-002)

* Severity: Medium
* Affected Function: `add_to_whitelist`
* Description: The `add_to_whitelist` function in the VotingEscrow contract does not have any restrictions, allowing any user to add an address to the whitelist.
* Exploit Code:
```solidity
pragma solidity ^0.8.0;
contract VotingEscrow {
    function add_to_whitelist(address _addr) public {
        whitelist[_addr] = true;
    }
}
```
* Fix Code:
```solidity
pragma solidity ^0.8.0;
contract VotingEscrow {
    address public owner;
    constructor() {
        owner = msg.sender;
    }
    modifier onlyOwner() {
        require(msg.sender == owner, 'Only the owner can call this function');
        _;
    }
    function add_to_whitelist(address _addr) public onlyOwner {
        whitelist[_addr] = true;
    }
}
```
* Test Code:
```solidity
pragma solidity ^0.8.0;
import 'forge-std/Test.sol';
contract VotingEscrowTest is Test {
    VotingEscrow public votingEscrow;
    address public user1;
    address public user2;
    function setUp() public {
        votingEscrow = new VotingEscrow();
        user1 = address(1);
        user2 = address(2);
    }
    function test_add_to_whitelist() public {
        vm.expectRevert('Only the owner can call this function');
        vm.prank(user1);
        votingEscrow.add_to_whitelist(user2);
    }
}
```

### 3. BaseEmissionsController Manipulation (VE-003)

* Severity: High
* Affected Function: `setBaseEmissionsController`
* Description: The combination of insecure base emissions controller setting in the Trust contract and unsecured unlocking in the VotingEscrow contract allows an attacker to manipulate the contract's state.
* Exploit Code:
```solidity
pragma solidity ^0.8.0;
contract Trust {
    function setBaseEmissionsController(address _controller) public {
        baseEmissionsController = _controller;
    }
}
contract VotingEscrow {
    function unlock() public {
        // attacker can call unlock to manipulate state
    }
}
```
* Fix Code:
```solidity
pragma solidity ^0.8.0;
contract Trust {
    function setBaseEmissionsController(address _controller) public onlyAuthorized {
        baseEmissionsController = _controller;
    }
}
contract VotingEscrow {
    function unlock() public onlyAuthorized {
        // only authorized users can call unlock
    }
}
```
* Test Code:
```solidity
pragma solidity ^0.8.0;
import 'forge-std/Test.sol';
contract TrustTest is Test {
    Trust trust;
    VotingEscrow votingEscrow;
    function setUp() public {
        trust = new Trust();
        votingEscrow = new VotingEscrow();
    }
    function testSetBaseEmissionsController() public {
        // Test that only authorized users can set base emissions controller
        address attacker = address(0x1);
        vm.expectRevert('Unauthorized');
        trust.setBaseEmissionsController(attacker);
    }
    function testUnlock() public {
        // Test that only authorized users can unlock
        votingEscrow.unlock();
        vm.expectRevert('Unauthorized');
    }
}
```
