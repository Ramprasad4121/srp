	* Severity: Medium

## Detailed Vulnerability Breakdown

### Unsecured use of reentrancy in VotingEscrow

```solidity
pragma solidity ^0.8.0;

contract VotingEscrow {
    function withdraw() public {
        // reentrancy vulnerability
    }
}
```

To fix this vulnerability, the contract should use a reentrancy lock to prevent recursive calls.

```solidity
pragma solidity ^0.8.0;

contract VotingEscrow {
    bool private _reentrancyLock;

    function withdraw() public {
        require(!_reentrancyLock, "Reentrancy lock is enabled");
        _reentrancyLock = true;
        // withdraw logic
        _reentrancyLock = false;
    }
}
```

### Unrestricted access to VotingEscrow whitelist

```solidity
pragma solidity ^0.8.0;

contract VotingEscrow {
    function add_to_whitelist(address _addr) public {
        // unrestricted access
    }
}
```

To fix this vulnerability, the contract should restrict access to the whitelist using a modifier or a separate authorization mechanism.

```solidity
pragma solidity ^0.8.0;

contract VotingEscrow {
    modifier onlyAuthorized {
        require(msg.sender == owner, "Only the owner can add to whitelist");
        _;
    }

    function add_to_whitelist(address _addr) public onlyAuthorized {
        // add to whitelist logic
    }
}
```

### Unsecured use of transferFrom in WrappedTrust

```solidity
pragma solidity ^0.8.0;

contract WrappedTrust {
    function transferFrom(address _from, address _to, uint256 _amount) public {
        // unsecured use of transferFrom
    }
}
```

To fix this vulnerability, the contract should check the return value of `transferFrom` to ensure that the transfer was successful.

```solidity
pragma solidity ^0.8.0;

contract WrappedTrust {
    function transferFrom(address _from, address _to, uint256 _amount) public {
        require(transferFrom(_from, _to, _amount), "Transfer failed");
    }
}
```
